This is the new route in app.py. Compare the two blocks line by line. The try and except have the same lines and the same message. Only the two places that talk to the user changed: input() became request.form, and print() became msg.

app.py · add after report() · excerpt 

@app.route("/bank", methods=["GET", "POST"])
def bank():
    memory_bank = MemoryBank(STORE, "set-1")
    msg = None
    if request.method == "POST":
        pid = request.form.get("problem_id", "").strip()
        try:
            memory_bank.add_problem(pid)
            msg = f"Added. Set holds {len(memory_bank.members())}."
        except SetFull as err:
            msg = f"Refused: {err}"
    return render_template("bank.html", members=memory_bank.members(),
                           message=msg)
